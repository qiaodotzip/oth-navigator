```ts
import { ApiError, ConflictError, isUnauthorized } from "../api/client";
import { toast } from "./toast";

/**
 * Whether sending the same request again could work. A refusal (a 4xx) will
 * be refused again; a request that never arrived, or a server that fell over
 * (a 5xx), may well go through the second time.
 */
export function worthRetrying(err: unknown): boolean {
  return !(err instanceof ApiError) || err.status >= 500;
}

/** What the server said, when it said something in words. */
export function serverWords(err: unknown): string | null {
  if (!(err instanceof ApiError)) return null;
  const said = (err.body as { error?: unknown } | null)?.error;
  return typeof said === "string" && said.trim() !== "" ? said.trim().replace(/\.$/, "") : null;
}

/**
 * Says a request failed, in words a teammate can act on (spec §6). Errors
 * stay until dismissed, and offer Retry only where trying again could work.
 *
 * `doing` finishes the sentence "Couldn't …": "save your change", "add L1.png".
 */
export function reportFailure(
  err: unknown,
  { doing, retry }: { doing: string; retry?: () => void | Promise<void> },
): void {
  // A 401 already opened the session-ended dialog; a toast would only contradict it.
  if (isUnauthorized(err)) return;
  if (err instanceof ConflictError) {
    toast({ tone: "error", message: `Couldn't ${doing}: someone changed it since. Reload to see the latest.` });
    return;
  }
  if (!worthRetrying(err)) {
    const said = serverWords(err) ?? `the server refused it (HTTP ${(err as ApiError).status})`;
    toast({ tone: "error", message: `Couldn't ${doing}: ${said}.` });
    return;
  }
  toast({
    tone: "error",
    message: err instanceof ApiError
      ? `Couldn't ${doing}: the server had a problem. Try again in a moment.`
      : `Couldn't ${doing}. Check your connection, then retry.`,
    action: retry ? { label: "Retry", run: retry } : undefined,
  });
}
```

Replace `packages/editor/src/ui/undo.ts`:

```ts
import { ConflictError, isUnauthorized, type Api, type EntityType } from "../api/client";
import type { Removal } from "../flow/removeLayer";
import type { Saver } from "../state/save";
import { worthRetrying } from "./failure";
import { toast } from "./toast";

/** What each part is called when an undo only half works. */
const PART: Partial<Record<EntityType, string>> = {
  portal: "the lift or stair",
  portalEndpoint: "its stop on this floor",
  room: "the space",
  pathway: "the pathway",
  entrance: "the entrance",
  placement: "the placed model",
  doorway: "the doorway",
  pathwayKind: "the walkway type",
};

const part = (r: Removal) => PART[r.type] ?? r.type;

/**
 * Deletes, then offers Undo, instead of asking "are you sure?" first.
 *
 * Every delete here is soft, so undoing is a restore at the version the delete
 * left (one more than it had). Restores run in reverse, so a lift comes back
 * before its last stop. If a teammate has touched something since, its restore
 * is refused and the toast says so — and says what did come back — rather
 * than overwriting what they did.
 */
export async function deleteWithUndo({
  api,
  saver,
  removals,
  label,
}: {
  api: Pick<Api, "restoreEntity">;
  saver: Pick<Saver, "remove" | "refresh">;
  removals: readonly Removal[];
  label: string;
}): Promise<boolean> {
  const done: Removal[] = [];
  for (const r of removals) {
    if (!(await saver.remove(r.type, r.id, r.version))) break;
    done.push(r);
  }
  if (done.length === 0) return false;

  // Kept across attempts, so Retry carries on from where the last try stopped
  // instead of restoring the first part twice.
  const pending = [...done].reverse();
  const restored: Removal[] = [];

  async function undo(): Promise<void> {
    try {
      while (pending.length > 0) {
        const next = pending[0]!;
        await api.restoreEntity(next.type, next.id, next.version + 1);
        restored.push(pending.shift()!);
      }
      toast({ tone: "success", message: `Restored ${label}` });
    } catch (err) {
      // A 401 means the session expired; the sign-in dialog over the page says so.
      if (isUnauthorized(err)) return;
      const stuck = pending[0]!;
      const conflict = err instanceof ConflictError;
      const retry = !conflict && worthRetrying(err) ? { label: "Retry", run: undo } : undefined;
      const why = conflict ? "was changed since" : retry ? "did not" : "was refused";
      const next = conflict
        ? "Reload to see the latest."
        : retry ? "Check your connection, then retry." : "Reload to see where it stands.";
      toast({
        tone: "error",
        action: retry,
        message: restored.length === 0
          ? conflict ? `Couldn't undo: ${label} was changed since. ${next}` : `Couldn't undo ${label}. ${next}`
          : `Couldn't fully undo ${label}: ${restored.map(part).join(" and ")} came back, but ${part(stuck)} ${why}. ${next}`,
      });
    } finally {
      // Whatever came back, or did not, the plan must show it: a half-done
      // undo is exactly when a stale screen misleads.
      await saver.refresh().catch(() => {});
    }
  }

  toast({ message: `Deleted ${label}`, action: { label: "Undo", run: undo } });
  return done.length === removals.length;
}
```

```ts
        return false;
```

```ts
      reportFailure(err, {
        doing: "load the map",
        retry: async () => {

        },
```

```ts
export type BuildKeyAction =
  | { kind: "tool"; tool: Tool }
  | { kind: "fit" }
  | { kind: "delete" }

 * Spec §11's keys, by lower-cased `KeyboardEvent.key`. Space is the canvas's
 * own (hold to pan), and so is Backspace while a shape is being drawn.
 */
export const BUILD_KEYS: Readonly<Record<string, BuildKeyAction>> = {
  v: { kind: "tool", tool: "select" },
  a: { kind: "tool", tool: "draw-area" },
  r: { kind: "tool", tool: "draw-room" },
  p: { kind: "tool", tool: "draw-pathway" },
  e: { kind: "tool", tool: "place-entrance" },
  l: { kind: "tool", tool: "place-portal" },
  f: { kind: "fit" },
  delete: { kind: "delete" },
  escape: { kind: "cancel" },
};

export type KeyInput = {
  key: string;
  ctrlKey?: boolean;
  metaKey?: boolean;
  altKey?: boolean;
  /** Where the key was pressed: a field that takes text keeps all its keys. */
  target?: EventTarget | null;
  /** True while a shape has points, when Backspace takes one back instead. */
  drafting?: boolean;
};

/** What a key press means in Build, or null when it means nothing here. */
export function buildKeyAction(input: KeyInput): BuildKeyAction | null {
  // Ctrl+R reloads, Cmd+L goes to the address bar: those belong to the browser.
  if (input.ctrlKey || input.metaKey || input.altKey) return null;
  if (isTyping(input.target)) return null;
  const key = input.key.toLowerCase();
  if (key === "delete" || key === "backspace") {
    // Mid-shape these take back a corner, which the canvas does; deleting the
    // selection from under a half-drawn shape would be a surprise.
    if (input.drafting) return null;
    // A Mac laptop has no Delete key: its Backspace is the one people press.
    return { kind: "delete" };
  }
  return BUILD_KEYS[key] ?? null;
}
```

```ts
/** `room:L1-a` for the address bar. Ids are slugs, so they never contain a colon. */
export function selectParam(selection: Selection): string {
  return selection ? `${selection.type}:${selection.id}` : "";
}

/** The floor a selectable thing is on, if it is still there. */
function levelOf(model: MapModel, type: Selectable, id: string): string | undefined {
  const on = (items: readonly { id: string; levelId: string; deletedAt?: string | null }[]) =>
    live(items).find(x => x.id === id)?.levelId;
  switch (type) {
    case "room": return on(model.rooms);
    case "pathway": return on(model.pathways);
    case "doorway": return on(model.doorways);
    case "entrance": return on(model.entrances);
    case "portalEndpoint": return on(model.portalEndpoints);
    case "placement": return on(model.placements);
  }
}

/**
 * What `?select=` points at, if it is still on this floor. A link to something
 * since deleted, or on another floor, selects nothing rather than something wrong.
 */
export function parseSelectParam(
  value: string | undefined, model: MapModel | null, levelId: string | null,
): Selection {
  if (!value || !model || !levelId) return null;
  const colon = value.indexOf(":");
  if (colon <= 0) return null;
  const type = value.slice(0, colon);
  const id = value.slice(colon + 1);
  if (!id || !TYPES.includes(type)) return null;
  return levelOf(model, type as Selectable, id) === levelId ? { type: type as Selectable, id } : null;
}
```

The rules, from spec §11's table and §5.3:
- **Floorplan** items are not validation issues: no image on the floor, or an image with no scale.
- Every other item is a validation issue filed under the step `buildStepForCode` names — the same mapping Publish's blockers link with, so a blocker and its to-do always agree.
- An issue with a level belongs to that floor. One with no level is building-wide (no entrance at all) and every floor lists it.
- A **place** issue belongs to the place's floor from `placeLevels` — its own level, else the footage it was filmed in. Binding is for confirmed units: `place.unbound` leaves out proposals (Review's) and walkways (built as paths). A place no floor claims is on no floor's list; Task 8's picker still offers it.
- A **lift or stair** with too few stops (`portal.endpoints.insufficient`, no level) is listed on each floor it stops on, selecting that stop; with no stops at all it is on every floor, as something to delete.

```ts
import { live, placeLevels, type MapModel, type ValidationIssue, type ValidationReport } from "@oth/model";
import type { LevelImage } from "../api/client";
import { BUILD_STEPS, buildStepForCode, type BuildStepId } from "../flow/floorProgress";
import type { Selection } from "../state/store";
import { imageForLevel } from "../state/underlay";

type Selectable = NonNullable<Selection>["type"];
const SELECTABLE: ReadonlySet<string> = new Set<Selectable>(
  ["room", "pathway", "doorway", "entrance", "portalEndpoint", "placement"],
);

/** What choosing a to-do does. */
export type TodoTarget =
  | { kind: "select"; selection: NonNullable<Selection> }
  /** A surveyed place has no shape of its own: choosing it asks which room it is. */
  | { kind: "place"; placeId: string }
  /** A lift or stair with no stops is on no floor and cannot be clicked — only deleted. */
  | { kind: "orphan-portal"; portalId: string; version: number }
  | { kind: "none" };

export type TodoItem = {
  key: string;
  step: BuildStepId;
  severity: "error" | "warning";
  code: string;
  message: string;
  target: TodoTarget;
};

export type StepTodos = Record<BuildStepId, TodoItem[]>;

export const EMPTY_TODOS: StepTodos = { floorplan: [], rooms: [], pathways: [], entrances: [], bind: [], dressing: [] };

const NONE: TodoTarget = { kind: "none" };

/**
 * What is left to do on each step of one floor — spec §11's "to-do from"
 * column. Built from the validation report, so a to-do and the Publish
 * blocker it answers always agree about which step fixes it.
 */
export function stepTodos(
  model: MapModel,
  report: ValidationReport | null,
  levelImages: readonly LevelImage[],
  levelId: string,
): StepTodos {
  const todos: StepTodos = { floorplan: [], rooms: [], pathways: [], entrances: [], bind: [], dressing: [] };

  // The first two things a floor needs are not validation issues at all.
  const image = imageForLevel([...levelImages], levelId);
  if (!image) {
    todos.floorplan.push({
      key: "floorplan.missing", step: "floorplan", severity: "warning", code: "floorplan.missing",
      message: "Add this floor's plan.", target: NONE,
    });
  } else if (image.metresPerPixel == null) {
    todos.floorplan.push({
      key: "floorplan.scale", step: "floorplan", severity: "warning", code: "floorplan.scale",
      message: `Set the scale of ${image.filename}.`, target: NONE,
    });
  }

  const floors = placeLevels(model);
  const places = new Map(live(model.places).map(p => [p.id, p]));
  const portals = new Map(live(model.portals).map(p => [p.id, p]));
  const stops = live(model.portalEndpoints);
  const seen = new Map<string, number>();

  for (const issue of report?.issues ?? []) {
    const step = buildStepForCode(issue.code);
    if (!step) continue;
    const target = targetOnThisFloor(issue);
    if (!target) continue;
    // Two issues can name the same thing (a place disagreeing with two neighbours).
    const base = `${issue.code}:${issue.entityId}`;
    const n = seen.get(base) ?? 0;
    seen.set(base, n + 1);
    todos[step].push({
      key: n === 0 ? base : `${base}#${n}`,
      step, severity: issue.severity, code: issue.code, message: issue.message, target,
    });
  }

  // Errors first: they are what stop Publish. Otherwise the report's own
  // order, which keeps like with like.
  for (const { id } of BUILD_STEPS) {
    todos[id] = [...todos[id].filter(t => t.severity === "error"), ...todos[id].filter(t => t.severity !== "error")];
  }
  return todos;

  /** Where on this floor an issue points, or null when it is not this floor's. */
  function targetOnThisFloor(issue: ValidationIssue): TodoTarget | null {
    if (issue.entityType === "place") {
      const place = places.get(issue.entityId);
      if (!place || floors.get(place.id) !== levelId) return null;
      // Binding is for confirmed units: a proposal is still Review's to decide,
      // and a walkway is built as a path, not bound to a room.
      if (issue.code === "place.unbound" && (place.status !== "confirmed" || place.pathwayKindId)) return null;
      return place.roomId
        ? { kind: "select", selection: { type: "room", id: place.roomId } }
        : { kind: "place", placeId: place.id };
    }
    if (issue.entityType === "portal") {
      const itsStops = stops.filter(e => e.portalId === issue.entityId);
      if (itsStops.length === 0) {
        // On no floor, so on every floor's list: the one place it can be deleted from.
        const portal = portals.get(issue.entityId);
        return portal ? { kind: "orphan-portal", portalId: portal.id, version: portal.version } : null;
      }
      const here = itsStops.find(e => e.levelId === levelId);
      return here ? { kind: "select", selection: { type: "portalEndpoint", id: here.id } } : null;
    }
    // No level means building-wide — no entrance anywhere — and any floor can fix that.
    if (issue.levelId !== undefined && issue.levelId !== levelId) return null;
    return SELECTABLE.has(issue.entityType)
      ? { kind: "select", selection: { type: issue.entityType as Selectable, id: issue.entityId } }
      : NONE;
  }
}
```

```ts
import { live, type BBox, type MapModel, type Vec2 } from "@oth/model";
import { fitViewport, type Viewport } from "../plan/viewport";
import type { Selection } from "../state/store";

/** Floor shown around a shape, so it is seen where it is. */
const MARGIN_M = 4;
/** Floor shown around a point: a lift stop filling the screen says nothing about where it is. */
const POINT_MARGIN_M = 12;
/** Room around the framed box on screen, in pixels. */
const PAD_PX = 60;

function around(points: readonly Vec2[], margin: number): BBox {
  const xs = points.map(p => p[0]);
  const ys = points.map(p => p[1]);
  return {
    minX: Math.min(...xs) - margin,
    minY: Math.min(...ys) - margin,
    maxX: Math.max(...xs) + margin,
    maxY: Math.max(...ys) + margin,
  };
}

/** The part of the floor to show for a selection, or null when it is not there. */
export function selectionBBox(model: MapModel | null, selection: Selection): BBox | null {
  if (!model || !selection) return null;
  const { type, id } = selection;
  const find = <T extends { id: string; deletedAt?: string | null }>(items: readonly T[]) =>
    live(items).find(x => x.id === id);
  switch (type) {
    case "room": {
      const r = find(model.rooms);
      return r ? around(r.footprint, MARGIN_M) : null;
    }
    case "pathway": {
      const p = find(model.pathways);
      return p ? around(p.polyline, MARGIN_M + p.widthM / 2) : null;
    }
    case "doorway": {
      const d = find(model.doorways);
      return d ? around(d.segment, POINT_MARGIN_M) : null;
    }
    case "entrance": {
      const e = find(model.entrances);
      return e ? around([e.point], POINT_MARGIN_M) : null;
    }
    case "portalEndpoint": {
      const e = find(model.portalEndpoints);
      return e ? around([e.point], POINT_MARGIN_M) : null;
    }
    case "placement": {
      const p = find(model.placements);
      return p ? around([[p.position[0], p.position[1]]], POINT_MARGIN_M) : null;
    }
  }
}

/** The view that frames a selection in the middle of the canvas, or null when there is nothing to frame. */
export function zoomViewportTo(
  model: MapModel | null, selection: Selection, widthPx: number, heightPx: number,
): Viewport | null {
  const box = selectionBBox(model, selection);
  return box ? fitViewport(box, widthPx, heightPx, PAD_PX) : null;
}
```

```ts
export type PlanOffset = { offsetX: number; offsetY: number };

    offsetY: toCm(start.offsetY - dyPx / pixelsPerMetre),
```

```ts
 * provisional metre per pixel from its saved offset, so it can be seen, and
 * measured on, before anyone has said how big it is. Never saved.
 */

/**
 * The image's extent in world metres. Image y grows downward while world y is
 * north, so the image hangs below its offset.
 */
function extent(image: LevelImage, cal: Calibration): BBox {
  const widthM = image.widthPx * cal.metresPerPixel;
  const heightM = image.heightPx * cal.metresPerPixel;
  return { minX: cal.offsetX, minY: cal.offsetY - heightM, maxX: cal.offsetX + widthM, maxY: cal.offsetY };
}
```

```tsx
import { useState } from "react";
import { BUILD_STEPS, type BuildStepId } from "../flow/floorProgress";
import { Button } from "../ui/Button";
import type { TodoItem } from "./todo";

/** Enough to act on, with the rest one click away: a floor can have 190 places to bind. */
const SHOWN_AT_FIRST = 50;

function Severity({ severity }: { severity: TodoItem["severity"] }) {
  return severity === "error" ? (
    <>
      <span aria-hidden className="w-3 shrink-0 font-semibold text-danger">✕</span>
      <span className="sr-only">Error: </span>
    </>
  ) : (
    <>
      <span aria-hidden className="w-3 shrink-0 font-semibold text-yours">!</span>
      <span className="sr-only">Warning: </span>
    </>
  );
}

/**
 * What is left on this step, each item a way to the thing it is about.
 * Errors come first: they are what stop Publish.
 */
export function StepTodo({
  step,
  items,
  onChoose,
  onDeleteOrphan,
}: {
  step: BuildStepId;
  items: readonly TodoItem[];
  onChoose: (item: TodoItem) => void;
  onDeleteOrphan: (item: TodoItem) => void;
}) {
  const [showAll, setShowAll] = useState(false);
  const { label } = BUILD_STEPS.find(s => s.id === step)!;
  const shown = showAll ? items : items.slice(0, SHOWN_AT_FIRST);

  return (
    <section aria-label="To do" className="flex min-h-0 flex-1 flex-col">
      <h2 className="px-3 pb-1 pt-3 text-xs font-semibold uppercase tracking-wide text-stone-500">
        To do{items.length > 0 ? ` · ${items.length}` : ""}
      </h2>
      {items.length === 0 ? (
        <p className="px-3 text-xs text-stone-600">
          <span aria-hidden className="text-confirmed">✓ </span>Nothing left to do on {label}.
        </p>
      ) : (
        <ul className="min-h-0 flex-1 overflow-auto px-2 pb-2">
          {shown.map(item => (
            <li key={item.key}>
              {item.target.kind === "select" || item.target.kind === "place" ? (
                <button
                  type="button"
                  onClick={() => onChoose(item)}
                  className="flex w-full gap-2 rounded px-2 py-1.5 text-left text-xs hover:bg-stone-100"
                >
                  <Severity severity={item.severity} />
                  <span>{item.message}</span>
                </button>
              ) : (
                <div className="flex items-start gap-2 px-2 py-1.5 text-xs">
                  <Severity severity={item.severity} />
                  <span className="flex-1">{item.message}</span>
                  {item.target.kind === "orphan-portal" && (
                    <Button size="sm" aria-label="Delete this lift" onClick={() => onDeleteOrphan(item)}>
                      Delete
                    </Button>
                  )}
                </div>
              )}
            </li>
          ))}
        </ul>
      )}
      {!showAll && items.length > SHOWN_AT_FIRST && (
        <div className="px-3 pb-3">

            Show all {items.length}
          </Button>
        </div>
      )}
    </section>

}
```

```ts
/** ISO A paper sizes in millimetres, long side first. */
export const PAPER_SIZES = {
  A0: [1189, 841],
  A1: [841, 594],
  A2: [594, 420],
  A3: [420, 297],
  A4: [297, 210],
} as const;

export type PaperSize = keyof typeof PAPER_SIZES;

/** Above this, the image is not the shape of the sheet — cropped, trimmed, or a reduction. */
export const SHAPE_TOLERANCE = 0.02;

/** "1:1500", "1 : 1500" or "1500" is 1500. Null for anything that is not a positive scale. */
export function parseRatio(text: string): number | null {
  const match = /^\s*(?:1\s*:\s*)?(\d+(?:\.\d+)?)\s*$/.exec(text);
  if (!match) return null;
  const ratio = Number(match[1]);
  return ratio > 0 ? ratio : null;
}

/**
 * A sheet's scale from its paper size and printed ratio.
 *
 * The right method for OTH's plans: their drawn scale bars are wrong by a
 * factor of three, but the sheet is exact — 420 mm across an A3 at 1:1500 is
 * 630 m, over 3000 px is 0.21 m/px. The image's long side is the paper's long
 * side, whichever way up it was scanned. `shapeMismatch` says how far the
 * image is from the paper's proportions; a cropped image would give a scale
 * that looks right and is not.
 */
export function sheetScale(
  paper: PaperSize, ratio: number, widthPx: number, heightPx: number,
): { metresPerPixel: number; shapeMismatch: number } {
  const [longMm, shortMm] = PAPER_SIZES[paper];
  const longPx = Math.max(widthPx, heightPx);
  const shortPx = Math.min(widthPx, heightPx);
  const paperShape = longMm / shortMm;
  return {
    metresPerPixel: (longMm * ratio) / 1000 / longPx,
    shapeMismatch: Math.abs(longPx / shortPx - paperShape) / paperShape,
  };
}
```

```tsx
/** Adds a floor's plan. The server retires the floor's old one as it stores this. */
export async function uploadPlan(api: Api, levelId: string, file: File): Promise<void> {
  let created: LevelImage;
  try {
    created = await api.uploadLevelImage(levelId, file);
  } catch (err) {
    reportFailure(err, { doing: `add ${file.name}`, retry: () => uploadPlan(api, levelId, file) });
    return;
  }
  const s = useEditorStore.getState();
  s.setLevelImages([...s.levelImages.filter(i => i.levelId !== levelId), created]);
  toast({ tone: "success", message: `Added ${file.name}. Now set its scale.` });
}

/**
 * Changes a plan's scale, place or opacity. The answer is the whole image, so
 * there is nothing else to reload: none of these changes any validation.
 */
export async function patchPlan(
  api: Api, image: LevelImage, patch: LevelImagePatch, words: { doing: string; done?: string },
): Promise<void> {
  let updated: LevelImage;
  try {
    updated = await api.patchLevelImage(image.id, image.version, patch);
  } catch (err) {
    reportFailure(err, { doing: words.doing, retry: () => patchPlan(api, image, patch, words) });
    return;
  }
  const s = useEditorStore.getState();
  s.setLevelImages(s.levelImages.map(i => (i.id === updated.id ? updated : i)));
  if (words.done) toast({ tone: "success", message: words.done });
}

/** The distance field for Measure scale, shown while the tool is in use. */
function MeasureScale({ api, image }: { api: Api; image: LevelImage }) {
  const draft = useEditorStore(s => s.calibrationDraft);
  const setTool = useEditorStore(s => s.setTool);
  const [metres, setMetres] = useState("");
  const ready = (draft?.length ?? 0) >= 2;

  function apply() {
    if (!draft || draft.length < 2) return;
    // The clicks are world points on the plan as drawn; the scale is about the
    // plan's own pixels, so convert them back through how it is drawn now.
    const shown = displayCalibration(image)!;
    try {
      const { metresPerPixel } = calibrationFromMeasurement(
        worldToImagePoint(shown, draft[0]!),
        worldToImagePoint(shown, draft[draft.length - 1]!),
        Number.parseFloat(metres),
      );
      void patchPlan(api, image, { metresPerPixel }, {
        doing: "set the scale",
        done: `Scale set to ${metresPerPixel.toFixed(4)} m per pixel.`,
      });
      setTool("select");
    } catch (err) {
      if (!(err instanceof CalibrationError)) throw err;
      toast({ tone: "error", message: `Couldn't set the scale: ${err.message}.` });
    }
  }

  return (
    <div className="space-y-2 rounded bg-stone-100 p-2">
      <label className="block text-xs text-stone-700">
        Real distance (metres)
        <input
          aria-label="Real distance in metres"
          inputMode="decimal"
          value={metres}
          onChange={e => setMetres(e.target.value)}
          onKeyDown={e => {
            if (e.key === "Enter") apply();
          }}
          className="mt-1 w-full rounded border border-stone-300 px-2 py-1 text-sm"
        />
      </label>
      <Button size="sm" variant="primary" disabled={!ready} onClick={apply}>
        Apply scale
      </Button>
    </div>
  );
}
```

```tsx
import { live, placeLevels, type Place, type Room } from "@oth/model";
import { useState } from "react";
import type { Api } from "../api/client";
import type { Saver } from "../state/save";
import { useEditorStore } from "../state/store";
import { matchPlaces } from "../survey/places";
import { Button } from "../ui/Button";
import { toast } from "../ui/toast";

/** Thumbnails are 240 px wide: the server makes them from the 4K frames. */
const THUMB_WIDTH = 240;

/** The first frame each place was seen in — the picture that says "this one". */
function firstFrames(sightings: readonly { placeId: string; keyframeId: string }[]): Map<string, string> {
  const frames = new Map<string, string>();
  for (const s of sightings) if (!frames.has(s.placeId)) frames.set(s.placeId, s.keyframeId);
  return frames;
}

/**
 * Binds a traced room to the place someone filmed there — where the geometry
 * and the footage, two independent descriptions of the building, meet. It is
 * also the first thing that moves a room off "unverified".
 */
export function BindPlaces({
  api,
  saver,
  room,
  pendingPlaceId,
  onBound,
}: {
  api: Api;
  saver: Saver;
  room: Room;
  pendingPlaceId: string | null;
  onBound: () => void;
}) {
  const model = useEditorStore(s => s.model);
  const [query, setQuery] = useState("");
  const [busy, setBusy] = useState(false);
  if (!model) return null;

  const bound = live(model.places).find(p => p.roomId === room.id);
  if (bound) {
    return (
      <p className="text-xs text-stone-700">
        <span aria-hidden className="text-confirmed">✓ </span>Bound to <span className="font-medium">{bound.nameEn}</span> from the survey.
      </p>
    );
  }

  const floors = placeLevels(model);
  // Confirmed units nothing has claimed: a proposal is still Review's to
  // decide, and a walkway is built as a path, not bound to a room.
  const open = live(model.places).filter(p => p.status === "confirmed" && !p.pathwayKindId && !p.roomId);
  const pendingFirst = (a: Place, b: Place) => Number(b.id === pendingPlaceId) - Number(a.id === pendingPlaceId);
  const here = matchPlaces(open.filter(p => floors.get(p.id) === room.levelId), query).sort(pendingFirst);
  const floorless = matchPlaces(open.filter(p => floors.get(p.id) === undefined), query).sort(pendingFirst);
  const sightings = live(model.sightings);
  const frames = firstFrames(sightings);
  const code = model.levels.find(l => l.id === room.levelId)?.code ?? "this floor";

  async function bind(place: Place) {
    setBusy(true);
    const evidence = sightings.filter(s => s.placeId === place.id).map(s => s.keyframeId);
    try {
      // The place takes the room's floor as well, so one filmed on footage
      // with no floor is filed where it was found.
      if (!(await saver.update("place", place.id, place.version, { roomId: room.id, levelId: room.levelId }))) return;
      await saver.update("room", room.id, room.version, {
        verification: { status: "photo", evidence, verifiedAt: new Date().toISOString() },
      });
      toast({ tone: "success", message: `Bound ${room.nameEn || room.id} to ${place.nameEn}.` });
      onBound();
    } finally {
      setBusy(false);
    }
  }

  const row = (p: Place) => {
    const frame = frames.get(p.id);
    return (
      <li key={p.id}>
        <button
          type="button"
          disabled={busy}
          aria-label={`Bind to ${p.nameEn}`}
          onClick={() => void bind(p)}
          className={`flex w-full items-center gap-2 rounded p-1 text-left hover:bg-stone-100 disabled:opacity-40 ${
            p.id === pendingPlaceId ? "ring-2 ring-accent-ring" : ""
          }`}
        >
          {frame ? (
            <img
              src={api.keyframeFileUrl(frame, THUMB_WIDTH)}
              alt=""
              loading="lazy"
              className="h-12 w-20 shrink-0 rounded bg-stone-200 object-cover"
            />
          ) : (
            <span className="flex h-12 w-20 shrink-0 items-center justify-center rounded bg-stone-100 text-xs text-stone-500">
              No frame
            </span>
          )}
          <span className="min-w-0">
            <span className="block truncate text-sm font-medium">{p.nameEn}</span>
            <span className="block text-xs text-stone-500">
              {p.category}
              {p.id === pendingPlaceId ? " · from your to-do" : ""}
            </span>
          </span>
        </button>
      </li>
    );
  };

  if (open.length === 0) {
    return (
      <p className="text-xs text-stone-600">
        No confirmed places are waiting for a room. Tag and review this floor's footage first.
      </p>
    );
  }

  return (
    <div className="space-y-2">
      <h3 className="text-xs font-semibold text-stone-700">Which place is this?</h3>
      <input
        aria-label="Find a place"
        placeholder="Find a place…"
        value={query}
        onChange={e => setQuery(e.target.value)}
        className="w-full rounded border border-stone-300 px-2 py-1 text-sm"
      />
      {here.length > 0 ? (
        <ul aria-label={`On ${code}`} className="space-y-1">{here.map(row)}</ul>
      ) : (
        <p className="text-xs text-stone-500">
          {query ? `Nothing on ${code} matches "${query}".` : `No places filmed on ${code} are waiting for a room.`}
        </p>
      )}
      {floorless.length > 0 && (
        <details>
          <summary className="cursor-pointer text-xs text-stone-600">Places with no floor ({floorless.length})</summary>
          <ul aria-label="With no floor" className="mt-1 space-y-1">{floorless.map(row)}</ul>
        </details>
      )}
    </div>
  );
}

/** The Bind places guide's line: which place you are finding a room for, and its frame. */
export function BindTools({
  api,
  pendingPlaceId,
  onCancel,
}: {
  api: Api;
  pendingPlaceId: string | null;
  onCancel: () => void;
}) {
  const model = useEditorStore(s => s.model);
  const place = pendingPlaceId ? live(model?.places ?? []).find(p => p.id === pendingPlaceId) : undefined;
  if (!model || !place) {
    return <p className="text-xs text-stone-600">Or pick a place from the to-do list, then click the room it is in.</p>;
  }
  const frame = live(model.sightings).find(s => s.placeId === place.id)?.keyframeId;

  return (
    <div role="status" className="flex items-start gap-2">
      {frame && (
        <img
          src={api.keyframeFileUrl(frame, THUMB_WIDTH)}
          alt={`Where ${place.nameEn} was filmed`}
          className="h-14 w-24 shrink-0 rounded bg-stone-200 object-cover"
        />
      )}
      <div className="flex-1 space-y-1 text-xs text-stone-700">
        <p>
          Click the room where you see <span className="font-medium">{place.nameEn}</span>.
        </p>
        <Button size="sm" variant="ghost" onClick={onCancel}>
          Cancel
        </Button>
      </div>
    </div>
  );
}
```

```tsx
import { useState } from "react";
import type { Api } from "../api/client";
import { lockFor } from "../collab/locks";
import { layerEntityType, levelLayers, type LayerItem } from "../flow/layers";
import { removalsFor } from "../flow/removeLayer";
import type { Saver } from "../state/save";
import { useEditorStore, type Selection } from "../state/store";
import { Button } from "../ui/Button";
import { deleteWithUndo } from "../ui/undo";

/**
 * Everything on the floor, by kind: find a thing by name, rename it, or
 * delete it — with Undo rather than "are you sure?".
 */
export function LayersDrawer({
  api,
  saver,
  onClose,
  onChoose,
}: {
  api: Api;
  saver: Saver;
  onClose: () => void;
  onChoose: (target: NonNullable<Selection>) => void;
}) {
  const model = useEditorStore(s => s.model);
  const levelId = useEditorStore(s => s.activeLevelId);
  const selection = useEditorStore(s => s.selection);
  const locks = useEditorStore(s => s.locks);
  const me = useEditorStore(s => s.user);
  const [editing, setEditing] = useState<string | null>(null);

  const rename = (item: LayerItem, nameEn: string) =>
    void saver.update(layerEntityType(item.type), item.id, item.version, { nameEn });

  const remove = (item: LayerItem) => {
    if (!model) return;
    void deleteWithUndo({ api, saver, removals: removalsFor(model, item), label: item.label });
  };

  return (
    <aside
      aria-label="Layers"
      className="absolute bottom-14 right-3 top-3 z-20 flex w-72 flex-col rounded border border-stone-200 bg-white shadow-lg"
    >
      <div className="flex items-center justify-between border-b border-stone-200 px-3 py-2">
        <h2 className="text-sm font-semibold">Layers</h2>
        <Button variant="ghost" size="sm" aria-label="Close layers" onClick={onClose}>
          ×
        </Button>
      </div>
      <div className="min-h-0 flex-1 overflow-auto p-2">
        {levelLayers(model, levelId).map(group => (
          <section key={group.type} className="mb-2">
            <h3 className="px-1 py-1 text-xs font-medium text-stone-500">
              {group.label} ({group.items.length})
            </h3>
            {group.items.length === 0 && <p className="px-1 pb-1 text-xs italic text-stone-500">none yet</p>}
            <ul>
              {group.items.map(item => {
                const target: NonNullable<Selection> = { type: layerEntityType(item.type), id: item.id };
                // Locks are held on the entity, so an area is looked up as the room it is.
                const heldBy = lockFor(locks, target, me?.id ?? null);
                const isSelected = selection?.id === item.id;
                return (
                  <li
                    key={item.id}
                    data-testid={`layer-${item.id}`}
                    className={`group flex items-center gap-1 rounded px-1 py-0.5 text-sm ${
                      isSelected ? "bg-accent-soft text-accent" : "hover:bg-stone-100"
                    }`}
                  >
                    {editing === item.id ? (
                      <input
                        aria-label={`Rename ${item.label}`}
                        autoFocus
                        defaultValue={item.label}
                        className="min-w-0 flex-1 rounded border border-stone-300 px-1 text-sm text-stone-900"
                        onBlur={e => {
                          setEditing(null);
                          const next = e.target.value.trim();
                          if (next && next !== item.label) rename(item, next);
                        }}
                        onKeyDown={e => {
                          if (e.key === "Enter") e.currentTarget.blur();
                          if (e.key === "Escape") setEditing(null);
                        }}
                      />
                    ) : (
                      <button
                        type="button"
                        title={item.id}
                        aria-current={isSelected ? "true" : undefined}
                        className="min-w-0 flex-1 truncate text-left"
                        onClick={() => onChoose(target)}
                      >
                        {item.label}
                      </button>
                    )}
                    {heldBy && (
                      <span className="shrink-0 text-xs" title={`${heldBy.lockedByName} is editing this`}>
                        🔒 {heldBy.lockedByName}
                      </span>
                    )}
                    {item.renamable && editing !== item.id && (
                      <button
                        type="button"
                        aria-label={`Rename ${item.label}`}
                        className="rounded px-1 text-xs text-stone-500 opacity-0 hover:text-stone-900 focus:opacity-100 group-hover:opacity-100"
                        onClick={() => setEditing(item.id)}
                      >
                        Rename
                      </button>
                    )}
                    <button
                      type="button"
                      aria-label={`Delete ${item.label}`}
                      className="rounded px-1 text-xs text-danger opacity-0 focus:opacity-100 group-hover:opacity-100"
                      onClick={() => remove(item)}
                    >
                      Delete
                    </button>
                  </li>
                );
              })}
            </ul>
          </section>
        ))}
      </div>
    </aside>
  );
}
```

```tsx
import { useState, type ReactNode } from "react";
import { Button } from "../ui/Button";
import { recall, remember } from "../ui/remember";

/** Whether the 3D preview is showing, remembered per browser: whoever likes it on keeps it on. */
export function usePreview(): [boolean, () => void] {
  const [on, setOn] = useState(() => recall("build.preview") === "on");
  const toggle = () => {
    const next = !on;
    setOn(next);
    remember("build.preview", next ? "on" : "off");
  };
  return [on, toggle];
}

export function PreviewToggle({ on, onToggle }: { on: boolean; onToggle: () => void }) {
  return (
    <Button size="sm" aria-pressed={on} onClick={onToggle}>
      3D
    </Button>
  );
}

/**
 * The preview floats over the plan rather than taking a column from it: it is
 * for checking what you drew, not for working in.
 */
export function PreviewPanel({ children }: { children: ReactNode }) {
  return (
    <section
      aria-label="3D preview"
      className="absolute bottom-14 left-3 z-20 h-64 w-96 overflow-hidden rounded border border-stone-200 bg-white shadow-lg"
    >
      {children}
    </section>
  );
}
```

```ts
import { live, ringCentroid, type MapModel } from "@oth/model";
import type { Selection } from "../state/store";

export type NewPlacement =
  | {
      ok: true;
      data: { id: string; levelId: string; assetId: string; roomId: string; position: [number, number, number] };
    }
  | { ok: false; reason: string };

/**
 * Where a model goes when it is placed on the selected room or detail. The
 * position is the offset its export used: zero for a level context mesh,
 * authored in building coordinates, and the centroid for a detail mesh,
 * authored about its own middle.
 */
export function newPlacement(
  model: MapModel, levelId: string, selection: Selection, assetId: string, now = Date.now(),
): NewPlacement {
  const anchor = selection?.type === "room" ? live(model.rooms).find(r => r.id === selection.id) : undefined;
  if (!anchor) return { ok: false, reason: "Select a room or detail first." };
  const position: [number, number, number] =
    anchor.kind === "detail" ? [...ringCentroid(anchor.footprint), 0] : [0, 0, 0];
  return { ok: true, data: { id: `${levelId}-place-${now}`, levelId, assetId, roomId: anchor.id, position } };
}
```

```ts
import { useEffect, useRef } from "react";
import type { BuildStepId } from "../flow/floorProgress";
import { useEditorStore } from "../state/store";
import { buildKeyAction } from "./keys";
import { stepOfTool } from "./steps";

/**
 * Build's keys, on one window listener.
 *
 * Delete and Escape are also the canvas's keys — a corner selected, a shape
 * half drawn — and which listener hears a key first depends on when each was
 * attached. So those two wait a tick and act only if nothing claimed the key
 * (preventDefault) in the meantime. The tool keys and F act at once.
 */
export function useBuildKeys(handlers: {
  onFit: () => void;
  onDelete: () => void;
  onStep: (step: BuildStepId) => void;
}): void {
  // The handlers change every render; the listener reads the latest, so it is attached once.
  const latest = useRef(handlers);
  latest.current = handlers;

  useEffect(() => {
    function onKey(event: KeyboardEvent) {
      const s = useEditorStore.getState();
      const action = buildKeyAction({
        key: event.key,
        ctrlKey: event.ctrlKey,
        metaKey: event.metaKey,
        altKey: event.altKey,
        target: event.target,
        drafting: (s.draft?.length ?? 0) > 0,
      });
      if (!action) return;

      if (action.kind === "tool") {
        s.setTool(action.tool);
        const owner = stepOfTool(action.tool);
        if (owner) latest.current.onStep(owner);
        return;
      }
      if (action.kind === "fit") {
        latest.current.onFit();
        return;
      }

      setTimeout(() => {
        if (event.defaultPrevented) return;
        if (action.kind === "delete") {
          latest.current.onDelete();
          return;
        }
        // Escape steps back one thing at a time: the shape in progress, then the tool, then the selection.
        const now = useEditorStore.getState();
        if (now.draft) now.clearDraft();
        else if (now.tool !== "select") now.setTool("select");
        else now.select(null);
      }, 0);
    }

    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, []);
}
```

```tsx
  floorplan: false, rooms: false, pathways: false, entrances: false, bind: false, dressing: false,

/** The address bar's own `?select=`, read fresh rather than from a render that may be a step behind it. */
function selectInAddress(): string {
  return new URLSearchParams(window.location.search).get("select") ?? "";
}

/**
 * One floor's Build tab (spec §11): the six steps across the top, the plan
 * with the step's guide floating over it, and on the right the selection above
 * the step's to-do list.
 *
 * It never loads the map — the workspace already has — and never returns early
 * without a user, so an expired session keeps the page behind the sign-in dialog.
 */
export function BuildWorkspace({ api, saver, route }: { api: Api; saver: Saver; route: FloorRoute }) {
  const model = useEditorStore(s => s.model);
  const report = useEditorStore(s => s.report);
  const levelImages = useEditorStore(s => s.levelImages);
  const levelId = useEditorStore(s => s.activeLevelId);
  const selection = useEditorStore(s => s.selection);
  const [canvasBox, canvasSize] = useElementSize();
  const [layersOpen, setLayersOpen] = useState(false);
  const [previewOn, togglePreview] = usePreview();
  const [pendingPlaceId, setPendingPlaceId] = useState<string | null>(null);

  const todos = useMemo(
    () => (model && levelId ? stepTodos(model, report, levelImages, levelId) : EMPTY_TODOS),
    [model, report, levelImages, levelId],
  );
  const done = useMemo(
    () => (model && levelId ? buildStepsDone(model, report, levelImages, levelId) : NOTHING_DONE),
    [model, report, levelImages, levelId],
  );
  // No step in the address: the first one not done yet — a new floor starts at its plan.
  const step: BuildStepId = parseStep(route.params.step) ?? BUILD_STEPS.find(s => !done[s.id])?.id ?? "floorplan";
  const counts = Object.fromEntries(BUILD_STEPS.map(s => [s.id, todos[s.id].length])) as Record<BuildStepId, number>;
  const image = imageForLevel(levelImages, levelId);
  const selectValue = route.params.select ?? "";

  const below = step === "floorplan" && levelId ? levelBelow(model, levelId) : null;
  const belowRings = useMemo(
    () => (model && below
      ? live(model.rooms).filter(r => r.levelId === below.id && r.kind !== "detail").map(r => r.footprint)
      : undefined),
    [model, below],
  );

  // A missing or unknown ?step= becomes a real one, so the address is always worth sending.
  useEffect(() => {
    if (model && levelId && route.params.step !== step) setParams({ step });
  }, [model, levelId, route.params.step, step]);

  function zoomTo(target: NonNullable<Selection>) {
    if (!canvasSize) return;
    const vp = zoomViewportTo(useEditorStore.getState().model, target, canvasSize.width, canvasSize.height);
    if (vp) useEditorStore.getState().setViewport(vp);
  }

  // The address → the selection: on arrival from a link, and on Back.
  const arrived = useRef<NonNullable<Selection> | null>(null);
  useEffect(() => {
    if (!model || !levelId) return;
    if (selectValue === selectParam(useEditorStore.getState().selection)) return;
    const wanted = parseSelectParam(selectValue, model, levelId);
    if (wanted) {
      useEditorStore.getState().select(wanted);
      arrived.current = wanted;
    } else if (selectValue) {
      // Since deleted, or on another floor: select nothing, and say so in the address too.
      setParams({ select: null });
    }
  }, [model, levelId, selectValue]);

  // Zoom to what a link pointed at once the canvas has a size — which is also
  // after the canvas's own first fit, so the zoom is not undone by it.
  useEffect(() => {
    if (arrived.current && canvasSize) {
      zoomTo(arrived.current);
      arrived.current = null;
    }
  });

  // The selection → the address, so what you are looking at can be sent to a teammate.
  useEffect(() => {
    const value = selectParam(useEditorStore.getState().selection);
    if (levelId && value !== selectInAddress()) setParams({ select: value || null });
  }, [selection, levelId]);

  function chooseStep(next: BuildStepId) {
    const s = useEditorStore.getState();
    // A tool from another step would leave the guide explaining one thing while the canvas does another.
    if (s.tool !== "select" && !STEP_GUIDES[next].tools.includes(s.tool)) s.setTool("select");
    setPendingPlaceId(null);
    setParams({ step: next });
  }

  function chooseTodo(item: TodoItem) {
    const s = useEditorStore.getState();
    s.setTool("select");
    if (item.target.kind === "select") {
      s.select(item.target.selection);
      zoomTo(item.target.selection);
    } else if (item.target.kind === "place") {
      s.select(null);
      setPendingPlaceId(item.target.placeId);
    }
  }

  function deleteOrphan(item: TodoItem) {
    if (item.target.kind !== "orphan-portal") return;
    void deleteWithUndo({
      api,
      saver,
      removals: [{ type: "portal", id: item.target.portalId, version: item.target.version }],
      label: "the lift with no stops",
    });
  }

  function fit() {
    if (!canvasSize) return;
    const s = useEditorStore.getState();
    const plan = displayBBox(imageForLevel(s.levelImages, s.activeLevelId));
    s.setViewport(fitViewport(contentBBox(ringsOnActiveLevel(s), plan), canvasSize.width, canvasSize.height));
  }

  async function deleteSelected() {
    const s = useEditorStore.getState();
    const found = selectionRemovals(s.model, s.activeLevelId, s.selection);
    if (!found) return;
    if (await deleteWithUndo({ api, saver, removals: found.removals, label: found.label })) {
      useEditorStore.getState().select(null);
    }
  }

  useBuildKeys({
    onFit: fit,
    onDelete: () => void deleteSelected(),
    onStep: next => {
      if (next !== step) chooseStep(next);
    },
  });

  function extrasFor(s: BuildStepId): ReactNode {
    switch (s) {
      case "floorplan":
        return <FloorplanTools api={api} />;
      case "rooms":
        return <RoomsTools saver={saver} />;
      case "pathways":
        return <PathwaysTools saver={saver} />;
      case "entrances":
        return null;
      case "bind":
        return <BindTools api={api} pendingPlaceId={pendingPlaceId} onCancel={() => setPendingPlaceId(null)} />;
      case "dressing":
        return <DressingTools api={api} saver={saver} />;
    }
  }

  return (
    <div className="flex h-full flex-col">
      <Stepper current={step} done={done} counts={counts} onChoose={chooseStep} />
      <LockBanner />
      <div className="flex min-h-0 flex-1">
        <div ref={canvasBox} className="relative min-w-0 flex-1 overflow-hidden bg-stone-100">
          {canvasSize && (
            <PlanCanvas
              width={canvasSize.width}
              height={canvasSize.height}
              imageUrl={api.levelImageFileUrl}
              showHint={false}
              belowRings={belowRings}
              onMoveUnderlay={offset => {
                if (image) void patchPlan(api, image, offset, { doing: "move the plan" });
              }}
              onReshapeRoom={(room, footprint) => void saver.update("room", room.id, room.version, { footprint })}
              onMovePlacement={(placement, position) =>
                void saver.update("placement", placement.id, placement.version, { position })}
              onCommitSpace={(kind, ring) => {
                if (levelId) void saver.create("room", newSpace(kind, levelId, ring));
              }}
              onCommitPathway={polyline => {
                if (levelId) void saver.create("pathway", newPathway(levelId, polyline));
              }}
              onCommitPoint={point => {
                const s = useEditorStore.getState();
                if (!s.model || !levelId) return;
                void commitPoint({ saver, model: s.model, tool: s.tool, levelId, point, say: message => toast({ message }) });
              }}
            />
          )}

          <StepGuide step={step} extras={extrasFor(step)} />
          {step === "floorplan" && !image && <PlanDropZone api={api} />}

          <div className="absolute bottom-3 left-3 z-20 flex gap-2">
            <Button size="sm" aria-expanded={layersOpen} onClick={() => setLayersOpen(o => !o)}>
              Layers
            </Button>
            <PreviewToggle on={previewOn} onToggle={togglePreview} />
          </div>

          {layersOpen && (
            <LayersDrawer
              api={api}
              saver={saver}
              onClose={() => setLayersOpen(false)}
              onChoose={target => {
                const s = useEditorStore.getState();
                s.setTool("select");
                s.select(target);
                zoomTo(target);
              }}
            />
          )}

          {previewOn && model && levelId && (
            <PreviewPanel>
              <MapScene
                model={model}
                activeLevelId={levelId}
                assetUrl={api.assetFileUrl}
                selectedId={selection?.type === "room" ? selection.id : null}
                onSelect={id => useEditorStore.getState().select({ type: "room", id })}
              />
            </PreviewPanel>
          )}
        </div>

        <aside aria-label="Selection and to-do" className="flex w-80 shrink-0 flex-col border-l border-stone-200 bg-white">
          <section aria-label="Selection" className="max-h-[55%] shrink-0 overflow-auto border-b border-stone-200">
            <h2 className="px-3 pt-3 text-xs font-semibold uppercase tracking-wide text-stone-500">Selection</h2>
            <SelectionInspector
              api={api}
              saver={saver}
              step={step}
              pendingPlaceId={pendingPlaceId}
              onBound={() => setPendingPlaceId(null)}
            />
          </section>
          <StepTodo key={step} step={step} items={todos[step]} onChoose={chooseTodo} onDeleteOrphan={deleteOrphan} />
        </aside>
      </div>
    </div>
  );
}
```

```js
    await page.getByText("Deleted Plan Check room").waitFor();
    await page.getByRole("button", { name: "Undo" }).click();
    await page.getByText("Restored Plan Check room").waitFor({ timeout: 10000 });
    await drawer.getByRole("button", { name: "Close layers" }).click();
    await shot("layers-undo");
  });

  await check("Bind places: a place's to-do asks for its room, with its frame", async () => {

    const first = todo.getByRole("button").first();
    const text = await first.innerText();
    await first.click();
    await guide("Bind places").getByText(/click the room where you see/i).waitFor();
    await shot("bind-pending");
    return text.slice(0, 80);
  });

  await check("Bind places: a room offers this floor's places, with thumbnails", async () => {
    await selectFromLayers("Plan Check room");
    const offered = page.getByRole("button", { name: /^Bind to / });
    await offered.first().waitFor();
    const thumbs = await page.locator('button[aria-label^="Bind to "] img').count();
    await shot("bind-offers");
    return `${await offered.count()} places offered, ${thumbs} with a frame`;
  });

  await check("Dressing: the context export and the models are offered", async () => {
    await goToStep("Dressing");
    await guide("Dressing").getByRole("button", { name: /Export context for Blender/ }).waitFor();
    await guide("Dressing").getByRole("button", { name: /Models/ }).click();
    await page.getByLabel("Upload a .glb").waitFor();
    await page.keyboard.press("Escape");
    await shot("dressing");
  });

  await check("3D: the preview opens as a floating panel, and closes", async () => {
    await page.getByRole("button", { name: "3D" }).click();
    await page.getByRole("region", { name: "3D preview" }).locator("canvas").waitFor({ timeout: 15000 });
    await shot("preview");
    await page.getByRole("button", { name: "3D" }).click();
  });

  await check("a link opens the same step and selection in a new tab", async () => {
    await goToStep("Areas & rooms");
    await selectFromLayers("Plan Check room");
    const url = page.url();
    const other = await page.context().newPage();
    await other.setViewportSize({ width: 1440, height: 900 });
    await other.goto(url);
    await other.getByRole("navigation", { name: "Build steps" }).waitFor();

    await other.close();
    if (name !== "Plan Check room") throw new Error(`opened with "${name}" selected`);
    return url.replace(BASE, "");

    await page.keyboard.press("f");
    await page.keyboard.press("Escape");
    await page.waitForTimeout(200);
```
